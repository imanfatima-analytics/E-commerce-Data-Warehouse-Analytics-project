
USE DATABASE ECOMMERCE_DW;
USE SCHEMA WAREHOUSE;

-- 1. Check missing values in the fact table
SELECT
    COALESCE(COUNT_IF(CUSTOMER_KEY IS NULL), 0) AS NULL_CUSTOMERS,
    COALESCE(COUNT_IF(PRODUCT_KEY IS NULL), 0) AS NULL_PRODUCTS,
    COALESCE(COUNT_IF(DATE_KEY IS NULL), 0) AS NULL_DATES,
    COALESCE(COUNT_IF(SALES_AMOUNT IS NULL), 0) AS NULL_SALES,
    COALESCE(COUNT_IF(PROFIT IS NULL), 0) AS NULL_PROFIT
FROM FACT_SALES;

-- 2. Check negative quantity, sales, and cost
SELECT
    COALESCE(COUNT_IF(QUANTITY < 0), 0) AS NEGATIVE_QUANTITY,
    COALESCE(COUNT_IF(SALES_AMOUNT < 0), 0) AS NEGATIVE_SALES,
    COALESCE(COUNT_IF(COST_AMOUNT < 0), 0) AS NEGATIVE_COST
FROM FACT_SALES;

-- 3. Validate profit calculations
SELECT
    COUNT(*) AS TOTAL_ROWS,
    COALESCE(
        COUNT_IF(
            ROUND(PROFIT, 2) !=
            ROUND(SALES_AMOUNT - COST_AMOUNT, 2)
        ), 0
    ) AS INVALID_PROFIT_ROWS
FROM FACT_SALES;

-- 4. Check missing order-status keys
SELECT
    COALESCE(COUNT_IF(ORDER_STATUS_KEY IS NULL), 0)
        AS NULL_STATUS_KEYS
FROM FACT_SALES;

-- 5. Identify orders with multiple fact records
-- Multiple records may be valid if the grain is order-line level.
SELECT
    ORDER_ID,
    COUNT(*) AS RECORD_COUNT
FROM FACT_SALES
GROUP BY ORDER_ID
HAVING COUNT(*) > 1
ORDER BY RECORD_COUNT DESC;
